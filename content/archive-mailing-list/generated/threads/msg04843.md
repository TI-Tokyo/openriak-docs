%%%-------------------------------------------------------------------
%%% @author &lt;roland.karls...@erlang-solutions.com&gt;
%%% @doc
%%% @end
%%%-------------------------------------------------------------------
-module(node).

-export([test_nvc/0,
         test_vc/0,
         test_vc_siblings/0,
         connect/0,
         create_bucket_nvc/0,
         create_bucket_vc/0,
         set_bucket_nvc/0,
         set_bucket_vc/0,
         write_first/0,
         write_second_nvc/0,
         write_second_vc/0,
         delete/0,
         read/0,
         buckets/0,
         keys/0,
         bucket_props/0]).

test_nvc() -&gt;
    create_bucket_nvc(),
    write_first(),
    write_second_nvc(),
    read().

test_vc() -&gt;
    create_bucket_vc(),
    write_first(),
    write_second_vc(),
    read().

test_vc_siblings() -&gt;
    create_bucket_vc(),
    write_first(),
    write_vc_siblings(),
    read().

connect() -&gt;
    {ok, Pid} = riakc_pb_socket:start_link(&quot;localhost&quot;, 8087),
    Pid.

create_bucket_nvc() -&gt;
    delete(),
    set_bucket_nvc().

set_bucket_nvc() -&gt;
    %% riakc_pb_socket:set_bucket/3 cannot be used. It only supports
    %% two settings: n_val and allow_mult.
    httpc:request(put,
                  {&quot;<a href="http://localhost:8098/riak/persons">http://localhost:8098/riak/persons</a>&quot;;,
                   [],
                   &quot;application/json&quot;,
                   &quot;{\&quot;props\&quot;:{\&quot;last_write_wins\&quot;:true,\&quot;allow_mult\&quot;:false}}&quot;},
                  [],
                  []).

create_bucket_vc() -&gt;
    delete(),
    set_bucket_vc().

set_bucket_vc() -&gt;
    %% riakc_pb_socket:set_bucket/3 cannot be used. It only supports
    %% two settings: n_val and allow_mult.
    httpc:request(put,
                  {&quot;<a href="http://localhost:8098/riak/persons">http://localhost:8098/riak/persons</a>&quot;;,
                   [],
                   &quot;application/json&quot;,
                   &quot;{\&quot;props\&quot;:{\&quot;last_write_wins\&quot;:false,\&quot;allow_mult\&quot;:true}}&quot;},
                  [],
                  []).

write_first() -&gt;
    Pid = connect(),
    Bucket = &lt;&lt;&quot;persons&quot;&gt;&gt;,
    Key = &lt;&lt;&quot;roland&quot;&gt;&gt;,
    Obj = riakc_obj:new(Bucket, Key, &lt;&lt;&quot;1953&quot;&gt;&gt;),
    ok = riakc_pb_socket:put(Pid, Obj),
    riakc_pb_socket:stop(Pid).

delete() -&gt;
    Pid = connect(),
    Bucket = &lt;&lt;&quot;persons&quot;&gt;&gt;,
    Key = &lt;&lt;&quot;roland&quot;&gt;&gt;,
    riakc_pb_socket:delete(Pid, Bucket, Key),
    %% In riak 1.0 you have to do some seconds sleep after a
    %% delete. Is this meant to be so?
    timer:sleep(5000),
    riakc_pb_socket:stop(Pid).

read() -&gt;
    Pid = connect(),
    Bucket = &lt;&lt;&quot;persons&quot;&gt;&gt;,
    Key = &lt;&lt;&quot;roland&quot;&gt;&gt;,
    {ok, Obj} = riakc_pb_socket:get(Pid, Bucket, Key),
    riakc_pb_socket:stop(Pid),
    Obj.

write_second_nvc() -&gt;
    Pid = connect(),
    Bucket = &lt;&lt;&quot;persons&quot;&gt;&gt;,
    Key = &lt;&lt;&quot;roland&quot;&gt;&gt;,
    Obj = riakc_obj:new(Bucket, Key, &lt;&lt;&quot;1954&quot;&gt;&gt;),
    ok = riakc_pb_socket:put(Pid, Obj),
    riakc_pb_socket:stop(Pid).

write_second_vc() -&gt;
    Pid = connect(),
    Bucket = &lt;&lt;&quot;persons&quot;&gt;&gt;,
    Key = &lt;&lt;&quot;roland&quot;&gt;&gt;,
    {ok, Obj1} = riakc_pb_socket:get(Pid, Bucket, Key),
    Obj2 = riakc_obj:update_value(Obj1, &lt;&lt;&quot;1954&quot;&gt;&gt;),
    ok = riakc_pb_socket:put(Pid, Obj2),
    riakc_pb_socket:stop(Pid).

write_vc_siblings() -&gt;
    Pid = connect(),
    Bucket = &lt;&lt;&quot;persons&quot;&gt;&gt;,
    Key = &lt;&lt;&quot;roland&quot;&gt;&gt;,
    {ok, Obj1} = riakc_pb_socket:get(Pid, Bucket, Key),
    Obj2 = riakc_obj:update_value(Obj1, &lt;&lt;&quot;1954&quot;&gt;&gt;),
    ok = riakc_pb_socket:put(Pid, Obj2),
    Obj3 = riakc_obj:update_value(Obj1, &lt;&lt;&quot;1955&quot;&gt;&gt;),
    ok = riakc_pb_socket:put(Pid, Obj3),
    riakc_pb_socket:stop(Pid).

buckets() -&gt;
    Pid = connect(),
    {ok, Buckets} = riakc_pb_socket:list_buckets(Pid),
    riakc_pb_socket:stop(Pid),
    Buckets.

keys() -&gt;
    Pid = connect(),
    Bucket = &lt;&lt;&quot;persons&quot;&gt;&gt;,
    {ok, Keys} = riakc_pb_socket:list_keys(Pid, Bucket),
    riakc_pb_socket:stop(Pid),
    Keys.

bucket_props() -&gt;
    Pid = connect(),
    Bucket = &lt;&lt;&quot;persons&quot;&gt;&gt;,
    %% NOTE: only returns allow_mult and n_val.
    {ok, Props} = riakc_pb_socket:get_bucket(Pid, Bucket),
    riakc_pb_socket:stop(Pid),
    Props.

Riak deletes by first writing a tombstone and then when all replicas are in
sync removing the object from the underlying key/value store.  We have made
some changes in 1.0.0 to increase the length of time the tombstones are
around when all nodes are up (in the Delete Changes section of the release
notes here
<a href="https://github.com/basho/riak/blob/1.0/RELEASE-NOTES.org">https://github.com/basho/riak/blob/1.0/RELEASE-NOTES.org</a> ).

If you just want to make sure the object is deleted, you could use the
deletedvclock option on the get request.  You will get a 3-tuple {error,
notfound, VClock} until the tombstone is removed, then a 2-tuple of {error,
notfound} when it has gone.

If you wish to rewrite the key, you can use the returned vclock on a new
object
  O = riakc_obj:set_vclock(riakc_obj:new(&lt;&lt;&quot;b&quot;&gt;&gt;,&lt;&lt;&lt;&quot;k&quot;&gt;&gt;,&lt;,&quot;v&quot;&gt;&gt;), VClock)

and it will overwrite the tombstone for you.
