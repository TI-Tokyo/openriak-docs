results_3 = Riak::MapReduce.new(client).

]]).

The FilterBuilder syntax I used follows, too:

filter { Riak::MapReduce::FilterBuilder.new }
filter.AND do
tokenize &quot;-&quot;, 1 do
eq &quot;test&quot;
end

ends_with &quot;123&quot;
end
