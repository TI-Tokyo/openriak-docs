{
   &quot;inputs&quot;:{
       &quot;bucket&quot;:&quot;data&quot;,
       &quot;index&quot;:&quot;numero_int&quot;,
       &quot;start&quot;:0,
       &quot;end&quot;:1212
   },
   &quot;query&quot;:[
    {&quot;map&quot;:
{&quot;language&quot;:&quot;erlang&quot;,&quot;module&quot;:&quot;riak_kv_mapreduce&quot;,&quot;function&quot;:&quot;map_object_value&quot;,&quot;arg&quot;:&quot;filter_notfound&quot;}
    },
       {&quot;reduce&quot;: {&quot;language&quot;:&quot;erlang&quot;,
&quot;module&quot;:&quot;riak_kv_mapreduce&quot;,&quot;function&quot;:&quot;reduce_count_inputs&quot;,&quot;arg&quot;:&quot;filter_notfound&quot;}}

    ], &quot;timeout&quot;: 9000
}

hope it helps,
-fyodor
