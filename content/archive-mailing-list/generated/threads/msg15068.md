This works for
me on Yosemite.

#!/usr/bin/env bash
MAXFILES=$(launchctl limit maxfiles | sed -e &#x27;s/[[:space:]]+/ /g&#x27; | xargs)
if [ &quot;$MAXFILES&quot; != &quot;maxfiles 65536 65536&quot; ]; then
  set -x
  sudo launchctl limit maxfiles 65536 65536
fi
set -x
ulimit -n 65536
riak start

I save the file to ~/bin/start-riak and use it instead of `riak start`.
