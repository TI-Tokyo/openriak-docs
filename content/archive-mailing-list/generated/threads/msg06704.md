I have a bash script put.sh:
----------------------------------------------------------------------------------------------------------
# upload file
#
# Bulk Upload:
# put.sh local-folder bucket-name prefix-inside-bucket
#
file=$1
bucket=$2
prefix=$3

if [ -d &quot;$file&quot; ]; then
  prefixRemove=$file find $file -type f -exec $0 {} $bucket $prefix \;
  exit 0
fi

if [ -z &quot;$type&quot; ]; then
type=`file -i $file | awk &#x27;{print $2}&#x27;`
fi

# escape
bucket=$(echo $bucket | sed &#x27;s/~/~~/g&#x27; | sed &#x27;s/\/\+/~/g&#x27; )

key=$file
key=$(echo $key | sed &#x27;s/^\.\/\+//g&#x27;)
if [ -n &quot;$prefixRemove&quot; ]; then
  prefixRemove=$(echo $prefixRemove | sed &#x27;s/\(\/\)/\\\1/g&#x27;)
  key=$(echo $key | eval &quot;sed &#x27;s/$prefixRemove//g&#x27;&quot;)
fi
key=$(echo $prefix$key | sed &#x27;s/~/~~/g&#x27; | sed &#x27;s/\/\+/~/g&#x27;)

#-H &quot;x-riak-index-date_int:`date +%s`&quot;
if [ -z &quot;$test&quot; ]; then
  curl -s -X PUT -T ${file} -H content-type:${type}
<a href="http://magic1:8098/buckets/${bucket}/keys/${key}">http://magic1:8098/buckets/${bucket}/keys/${key}</a>
fi
echo <a href="http://magic1:8098/buckets/${bucket}/keys/${key}">http://magic1:8098/buckets/${bucket}/keys/${key}</a>
----------------------------------------------------------------------------------------------------------

----------------------------------------------------------------------------------------------------------
# upload file
#
# Bulk Upload:
# put.sh local-folder bucket-name prefix-inside-bucket
#
file=$1
bucket=$2
prefix=$3

if [ -d &quot;$file&quot; ]; then
prefixRemove=$file find $file -type f -exec $0 {} $bucket $prefix \;
exit 0
fi

if [ -z &quot;$type&quot; ]; then
type=`file -i $file | awk &#x27;{print $2}&#x27;`
fi

# escape
bucket=$(echo $bucket | sed &#x27;s/~/~~/g&#x27; | sed &#x27;s/\/\+/~/g&#x27; )

key=$file
key=$(echo $key | sed &#x27;s/^\.\/\+//g&#x27;)
if [ -n &quot;$prefixRemove&quot; ]; then
prefixRemove=$(echo $prefixRemove | sed &#x27;s/\(\/\)/\\\1/g&#x27;)
key=$(echo $key | eval &quot;sed &#x27;s/$prefixRemove//g&#x27;&quot;)
fi
key=$(echo $prefix$key | sed &#x27;s/~/~~/g&#x27; | sed &#x27;s/\/\+/~/g&#x27;)

#-H &quot;x-riak-index-date_int:`date +%s`&quot;
if [ -z &quot;$test&quot; ]; then
curl -s -X PUT -T ${file} -H content-type:${type}
<a href="http://magic1:8098/buckets/${bucket}/keys/${key}">http://magic1:8098/buckets/${bucket}/keys/${key}</a>
fi
echo <a href="http://magic1:8098/buckets/${bucket}/keys/${key}">http://magic1:8098/buckets/${bucket}/keys/${key}</a>

----------------------------------------------------------------------------------------------------------
