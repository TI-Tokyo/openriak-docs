{{- $context := partial "product-context.html" . -}} {{- $versionData := index hugo.Data.page_provenance $context.id $context.version -}} {{- $versionRoot := printf "%s/%s" $context.id $context.version -}} {{- $prefix := printf "%s/" $versionRoot -}} {{- $contentPath := strings.TrimPrefix "/" .Path | strings.TrimSuffix "/" -}} {{- $pageKey := "" -}} {{- if and (ne $contentPath $context.id) (ne $contentPath $versionRoot) -}} {{- $pageKey = strings.TrimPrefix $prefix $contentPath -}} {{- end -}} {{- $entry := index $versionData $pageKey -}} {{- if not $entry -}} {{- if hugo.IsServer -}} {{- $entry = dict "status" "new" "since" $context.version -}} {{- else -}} {{- errorf "missing page provenance for product=%s version=%s page=%s" $context.id $context.version $pageKey -}} {{- end -}} {{- end -}} {{- if in (slice "downloads" "downloads/for-docker") $pageKey -}} {{/* Changed Markdown is new; unchanged Markdown still lists downloads updated for this release. */}} {{- $status := cond (eq $entry.status "inherited") "updated" "new" -}} {{- $entry = dict "status" $status "since" $context.version -}} {{- end -}} {{- return $entry -}}
